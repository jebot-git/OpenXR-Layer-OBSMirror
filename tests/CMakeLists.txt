cmake_minimum_required(VERSION 3.16)
project(OBSMirrorVulkanTests LANGUAGES CXX)
find_package(Vulkan REQUIRED)
enable_testing()
add_executable(vulkan_readback_test vulkan_readback_test.cpp
    ../XR_APILAYER_NOVENDOR_OBSMirror/vulkan_readback.cpp)
target_include_directories(vulkan_readback_test PRIVATE ../XR_APILAYER_NOVENDOR_OBSMirror)
target_compile_features(vulkan_readback_test PRIVATE cxx_std_17)
target_link_libraries(vulkan_readback_test PRIVATE Vulkan::Vulkan)
if(MSVC)
    target_compile_options(vulkan_readback_test PRIVATE /W4)
else()
    target_compile_options(vulkan_readback_test PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
endif()
add_test(NAME vulkan_readback COMMAND vulkan_readback_test)
